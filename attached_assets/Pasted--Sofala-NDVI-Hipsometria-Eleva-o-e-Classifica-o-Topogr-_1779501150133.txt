// Sofala — NDVI, Hipsometria, Elevação e Classificação Topográfica
// ================================================================

// ---------- 0. PARÂMETROS ----------
var EXPORT_SCALE = 30;         // resolução de exportação em metros (30 m para GLO-30)
var EXPORT_TO_DRIVE = true;    // true -> Export to Drive, false -> desativa exports
// Parâmetros do Landsat
var START_DATE = '2023-01-01'; // Data de início para buscar imagens Landsat
var END_DATE = '2024-01-01';   // Data de fim

// ---------- 1. ÁREA DE ESTUDO (SOFALA) ----------
var adminLevel3 = ee.FeatureCollection("projects/imersaogee-476916/assets/admin");
var sofala = adminLevel3.filter(ee.Filter.eq("NAME_1", "Sofala")).geometry();
Map.centerObject(sofala, 7);


// ---------- 2. DEM (COPERNICUS GLO-30) e ELEVAÇÃO ----------
var dem = ee.ImageCollection("COPERNICUS/DEM/GLO30")
  .select("DEM")
  .filterBounds(sofala)
  .mosaic()
  .clip(sofala);

// DEM Suavizado (Elevação)
var demSmooth = dem.focal_mean(3, 'square', 'meters').rename('DEM');

// Estatísticas para normalização da Hipsometria
var demStats = demSmooth.reduceRegion({
  reducer: ee.Reducer.minMax(),
  geometry: sofala,
  scale: EXPORT_SCALE,
  bestEffort: true,
  maxPixels: 1e9
});

var demMin = ee.Number(demStats.get('DEM_min'));
var demMax = ee.Number(demStats.get('DEM_max'));


// ---------- 3. RIOS (VECTOR) e rasterização ----------
var rivers = ee.FeatureCollection("WWF/HydroSHEDS/v1/FreeFlowingRivers")
  .filterBounds(sofala);

// raster dos rios (valor 1) - Usado na Classificação Topográfica
var riversRaster = ee.Image().byte().paint(rivers, 1).clip(sofala).rename('water');
Map.addLayer(riversRaster.selfMask(), {palette: ['3366ff']}, 'Rios (Vector/Raster)', false);


// ----------------------------------------------------------------
// ---------- 4. NDVI (Landsat 8) ----------
// ----------------------------------------------------------------
var l8 = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2')
  .filterBounds(sofala)
  .filterDate(START_DATE, END_DATE)
  .filter(ee.Filter.lt('CLOUD_COVER', 20)); 

// Função de máscara de nuvem
function applyCloudMask(image) {
  var qa = image.select('QA_PIXEL');
  var cloudMask = qa.bitwiseAnd(1 << 3).eq(0) // Nuvens (Cloud)
                    .and(qa.bitwiseAnd(1 << 5).eq(0)); // Cirrus
  return image.updateMask(cloudMask);
}

// Composite mediano sem nuvens
var l8Composite = l8.map(applyCloudMask).median().clip(sofala);

// NDVI: (NIR - Red) / (NIR + Red)
var ndviImage = l8Composite.normalizedDifference(['SR_B5', 'SR_B4']).rename('NDVI');
var ndviPalette = ['ffffff', 'ce7e45', 'df923d', 'f1b555', 'fcd163', '99b718', '74a901', '669100', '528d01', '3e8200', '207401', '056201', '004c00', '023b01', '012e01', '011d01', '011301'];
Map.addLayer(ndviImage, {min: 0.0, max: 0.8, palette: ndviPalette}, 'NDVI (Vegetação)', false);


// ----------------------------------------------------------------
// ---------- 5. HIPSOMETRIA (Elevação Colorida Contínua) ----------
// ----------------------------------------------------------------

// Normaliza (0-1)
var demNorm = demSmooth.subtract(demMin).divide(demMax.subtract(demMin)).clamp(0,1).rename('dem_norm');
// Paleta hipsométrica (podes ajustar)
var hipPalette = [
  '003f5c','2f4b7c','465881','5a6a86','6f7b92',
  '8c9aa6','b5c0c8','f7b267','f08a5d','e4572e','b71c1c'
];
var hips = demNorm.visualize({min:0, max:1, palette: hipPalette}).clip(sofala);
Map.addLayer(hips, {}, 'Hipsometria (Elevação Colorida)', false);


// ----------------------------------------------------------------
// ---------- 6. CLASSIFICAÇÃO TOPOGRÁFICA (classes) ----------
// ----------------------------------------------------------------
// Paleta de cores para as 6 classes
var paletteClass = ['d0f0ff','a0e060','ffff66','ffb366','ff6666','3366ff'];

// Definimos a classificação baseada no DEM, usando 'riversRaster' como a classe 6 (Água)
var classified = demSmooth.expression(
  "(water==1) ? 6" + // Classe 6: Água (Rios)
  ": (DEM < 5) ? 1" + // Classe 1: < 5m
  ": (DEM >=5 && DEM < 10) ? 2" + // Classe 2: 5m-10m
  ": (DEM >=10 && DEM < 30) ? 3" + // Classe 3: 10m-30m
  ": (DEM >=30 && DEM < 60) ? 4" + // Classe 4: 30m-60m
  ": 5", // Classe 5: > 60m
  { 'DEM': demSmooth, 'water': riversRaster.unmask(0) }
).rename('topo_class').clip(sofala);

Map.addLayer(classified, {min:1, max:6, palette: paletteClass}, 'Topografia - classes');


// ----------------------------------------------------------------
// ---------- 7. LEGENDA (Classificação Topográfica - Estilo Bangladesh) ----------
// ----------------------------------------------------------------

// Definição das classes e cores (usando paletteClass)
var classNames = [
  'Planície / < 5 m',           // Classe 1
  'Planície Baixa / 5 m - 10 m', // Classe 2
  'Planície Média / 10 m - 30 m', // Classe 3
  'Colinas Baixas / 30 m - 60 m', // Classe 4
  'Colinas Altas / > 60 m',       // Classe 5
  'Água & Rios',                // Classe 6
];
// Note: A ordem dos nomes deve corresponder à ordem da paletteClass:
// paletteClass[0] -> classNames[0], etc.

function addLegend() {
  var legend = ui.Panel({style:{position:'bottom-left',padding:'8px 15px', backgroundColor:'white'}});
  
  // Título da legenda
  legend.add(ui.Label({
    value:'Classificação Topográfica (m) - Zambézia',
    style:{fontWeight:'bold',fontSize:'16px',margin:'0 0 6px 0'}
  }));
  
  // Adiciona os itens da legenda usando o loop 'for'
  for (var i=0; i < classNames.length; i++){
    var colorBox = ui.Label({
      style:{
        backgroundColor:paletteClass[i], // Usa a cor da paletteClass
        padding:'8px',
        margin:'0 6px 4px 0',
        border:'1px solid #000'
      }
    });
    var desc = ui.Label({value:classNames[i],style:{margin:'0 0 4px 0'}});
    
    // Adiciona o painel horizontal (cor + descrição)
    legend.add(ui.Panel({widgets:[colorBox,desc],layout:ui.Panel.Layout.Flow('horizontal')}));
  }

  // Informação de fonte (pequena, itálico)
  legend.add(ui.Label({
    value:'Fonte: COPERNICUS DEM (GLO-30) & HydroSHEDS',
    style:{fontSize:'10px',fontStyle:'italic',margin:'6px 0 0 0'}
  }));

  Map.add(legend);
}
addLegend();


// ---------- 8. EXPORTAÇÃO GEO_TIFFS (inalterada) ----------
var regionGeom = sofala.bounds().buffer(1000); 

if (EXPORT_TO_DRIVE) {
  // 1) Exportar DEM suavizado (Elevação)
  Export.image.toDrive({
    image: demSmooth.toFloat(),
    description: 'Sofala_DEM_GLO30',
    folder: 'GEE_Exports',
    fileNamePrefix: 'Sofala_DEM_GLO30',
    region: regionGeom,
    scale: EXPORT_SCALE,
    maxPixels: 1e13
  });

  // 2) Exportar NDVI
  Export.image.toDrive({
    image: ndviImage.toFloat(),
    description: 'Sofala_NDVI',
    folder: 'GEE_Exports',
    fileNamePrefix: 'Sofala_NDVI',
    region: regionGeom,
    scale: EXPORT_SCALE,
    maxPixels: 1e13
  });

  // 3) Exportar Hipsometria (normalizada)
  Export.image.toDrive({
    image: demNorm.toFloat(),
    description: 'Sofala_Hipsometria_Normalizada',
    folder: 'GEE_Exports',
    fileNamePrefix: 'Sofala_Hipsometria',
    region: regionGeom,
    scale: EXPORT_SCALE,
    maxPixels: 1e13
  });

  // 4) Exportar a classificação topográfica
  Export.image.toDrive({
    image: classified.toInt(),
    description: 'Sofala_TopoClass',
    folder: 'GEE_Exports',
    fileNamePrefix: 'Sofala_TopoClass',
    region: regionGeom,
    scale: EXPORT_SCALE,
    maxPixels: 1e13
  });
} // end export


// ---------- 9. INFORMAÇÕES E NOTAS ----------
print('DEM min/max (m):', demMin, demMax);
print('Landsat 8 Source:', l8.size(), 'imagens Landsat 8 usadas no composite.');
print('Observações:');
print('- Exports -> Drive: verifica a pasta GEE_Exports.');
